## 📝 Passo a Passo da Implementação

### 1. Configurar Variáveis de Ambiente

Crie o arquivo `.env` na raiz do projeto:

```env
VITE_SUPABASE_PROJECT_ID="seu-project-id"
VITE_SUPABASE_PUBLISHABLE_KEY="sua-chave-publica-anon"
VITE_SUPABASE_URL="https://seu-project-id.supabase.co"
```

**Como obter as credenciais:**
1. Acesse [supabase.com](https://supabase.com) e faça login
2. Selecione seu projeto
3. Vá em **Settings > API**
4. Copie:
   - **Project URL** → `VITE_SUPABASE_URL`
   - **anon/public key** → `VITE_SUPABASE_PUBLISHABLE_KEY`
   - **Project Reference ID** → `VITE_SUPABASE_PROJECT_ID`

⚠️ **IMPORTANTE**: Adicione `.env` ao `.gitignore` para não commitar credenciais!

### 2. Criar Cliente Supabase com Configurações Otimizadas

Crie `src/integrations/supabase/client.ts`:

```typescript
// This file is automatically generated. Do not edit it directly.
import { createClient } from '@supabase/supabase-js';
import type { Database } from './types';

const SUPABASE_URL = import.meta.env.VITE_SUPABASE_URL;
const SUPABASE_PUBLISHABLE_KEY = import.meta.env.VITE_SUPABASE_PUBLISHABLE_KEY;

// Import the supabase client like this:
// import { supabase } from "@/integrations/supabase/client";

export const supabase = createClient<Database>(SUPABASE_URL, SUPABASE_PUBLISHABLE_KEY, {
  auth: {
    storage: localStorage,           // Persistir sessão no localStorage
    persistSession: true,            // Manter sessão entre reloads
    autoRefreshToken: true,          // Renovar token automaticamente
  }
});
```

**Configurações Críticas:**
- `storage: localStorage` - Persiste a sessão do usuário
- `persistSession: true` - Mantém o usuário logado após reload
- `autoRefreshToken: true` - Renova tokens automaticamente antes de expirar

### 4. Criar Sistema de Gerenciamento de Sessão

Crie `src/lib/auth.ts`:

```typescript
import { supabase } from "@/integrations/supabase/client";
import type { Session, User } from "@supabase/supabase-js";

let currentSession: Session | null = null;
let sessionPromise: Promise<Session | null> | null = null;

// Listener para mudanças de autenticação
supabase.auth.onAuthStateChange((_, session) => {
  currentSession = session;
  sessionPromise = null;
});

// Obter sessão de forma eficiente (com cache)
export async function getSessionOnce(): Promise<Session | null> {
  if (currentSession) return currentSession;
  if (sessionPromise) return sessionPromise;

  sessionPromise = supabase.auth.getSession().then(({ data: { session } }) => {
    currentSession = session;
    sessionPromise = null;
    return session;
  });

  return sessionPromise;
}

// Obter usuário atual
export async function getUserOnce(): Promise<User | null> {
  const session = await getSessionOnce();
  return session?.user ?? null;
}
```

**Por que isso é importante:**
- Evita múltiplas chamadas desnecessárias ao Supabase
- Cache inteligente da sessão
- Previne race conditions
- Melhora performance significativamente

### 5. Criar Hook de Autenticação React

Crie `src/hooks/useAuth.tsx`:

```typescript
import { useEffect, useState } from "react";
import { supabase } from "@/integrations/supabase/client";
import { getSessionOnce } from "@/lib/auth";
import type { User, Session } from "@supabase/supabase-js";

export function useAuth() {
  const [user, setUser] = useState<User | null>(null);
  const [session, setSession] = useState<Session | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // Listener para mudanças de autenticação
    const { data: { subscription } } = supabase.auth.onAuthStateChange(
      (_event, session) => {
        setSession(session);
        setUser(session?.user ?? null);
        setLoading(false);
      }
    );

    // Carregar sessão inicial
    getSessionOnce().then((session) => {
      setSession(session);
      setUser(session?.user ?? null);
      setLoading(false);
    });

    // Cleanup
    return () => subscription.unsubscribe();
  }, []);

  const signIn = async (email: string, password: string) => {
    const { error } = await supabase.auth.signInWithPassword({ email, password });
    return { error };
  };

  const signUp = async (email: string, password: string) => {
    const { error } = await supabase.auth.signUp({
      email,
      password,
      options: { emailRedirectTo: window.location.origin },
    });
    return { error };
  };

  const signOut = async () => {
    await supabase.auth.signOut();
  };

  return { user, session, loading, signIn, signUp, signOut };
}
```

### 8. Criar Hook Customizado para Operações de Banco

Exemplo de hook para gerenciar pedidos (`src/hooks/usePedidos.ts`):

```typescript
import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import { supabase } from "@/integrations/supabase/client";
import { getSessionOnce } from "@/lib/auth";

export function usePedidos() {
  const queryClient = useQueryClient();

  // Buscar pedidos
  const { data: pedidos = [], isLoading } = useQuery({
    queryKey: ["pedidos"],
    queryFn: async () => {
      const session = await getSessionOnce();
      if (!session) throw new Error("Não autenticado");

      const { data, error } = await supabase
        .from("pedidos")
        .select(`
          *,
          clientes(nome),
          transportadoras(razao_social)
        `)
        .order("data_pedido", { ascending: false });

      if (error) throw error;
      return data;
    },
  });

  // Criar pedido
  const createMutation = useMutation({
    mutationFn: async (novoPedido) => {
      const { data, error } = await supabase
        .from("pedidos")
        .insert(novoPedido)
        .select()
        .single();

      if (error) throw error;
      return data;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["pedidos"] });
    },
  });

  return {
    pedidos,
    isLoading,
    createPedido: createMutation.mutate,
    isCreating: createMutation.isPending,
  };
}
```

**Padrões importantes:**
- Sempre verificar sessão antes de queries
- Usar `getSessionOnce()` para eficiência
- Invalidar queries após mutations
- Tratar erros adequadamente

### 9. Configurar Row Level Security (RLS)

No Supabase Dashboard ou via SQL:

```sql
-- Habilitar RLS na tabela
ALTER TABLE pedidos ENABLE ROW LEVEL SECURITY;

-- Política: Usuários autenticados podem ler
CREATE POLICY "authenticated_read_pedidos"
  ON pedidos
  FOR SELECT
  TO authenticated
  USING (true);

-- Política: Usuários autenticados podem inserir
CREATE POLICY "authenticated_insert_pedidos"
  ON pedidos
  FOR INSERT
  TO authenticated
  WITH CHECK (true);

-- Política: Usuários autenticados podem atualizar
CREATE POLICY "authenticated_update_pedidos"
  ON pedidos
  FOR UPDATE
  TO authenticated
  USING (true)
  WITH CHECK (true);

-- Política: Apenas admin pode deletar
CREATE POLICY "admin_delete_pedidos"
  ON pedidos
  FOR DELETE
  TO authenticated
  USING (auth.jwt() ->> 'role' = 'admin');
```

## 🔒 Segurança e Boas Práticas

### 1. Variáveis de Ambiente

✅ **FAZER:**
- Usar `VITE_` prefix para variáveis expostas ao cliente
- Adicionar `.env` ao `.gitignore`
- Criar `.env.example` com placeholders
- Usar apenas a chave `anon/public` no frontend

❌ **NÃO FAZER:**
- Commitar `.env` no Git
- Usar `service_role` key no frontend
- Hardcodar credenciais no código

### 2. Row Level Security (RLS)

✅ **SEMPRE:**
- Habilitar RLS em todas as tabelas
- Criar políticas específicas por operação (SELECT, INSERT, UPDATE, DELETE)
- Testar políticas com diferentes usuários
- Usar `auth.uid()` para filtrar dados por usuário

### 3. Autenticação

✅ **IMPLEMENTAR:**
- Verificação de sessão antes de queries sensíveis
- Auto-refresh de tokens
- Logout em caso de token inválido
- Redirecionamento para login quando não autenticado

### 4. Queries Eficientes

✅ **OTIMIZAR:**
- Usar `select()` específico ao invés de `select('*')`
- Implementar paginação para listas grandes
- Usar índices no banco de dados
- Cache com React Query

## 🐛 Troubleshooting

### Erro: "Invalid API key"

**Causa:** Chave incorreta ou não configurada

**Solução:**
1. Verificar se `.env` existe e está correto
2. Confirmar que está usando a chave `anon/public`
3. Reiniciar o servidor de desenvolvimento
4. Verificar se o prefixo `VITE_` está presente

### Erro: "Row Level Security policy violation"

**Causa:** Usuário não tem permissão para acessar dados

**Solução:**
1. Verificar se o usuário está autenticado
2. Revisar políticas RLS da tabela
3. Testar políticas no SQL Editor do Supabase
4. Verificar se `auth.uid()` está retornando valor correto

### Erro: "Table doesn't exist"

**Causa:** Migrações não foram aplicadas

**Solução:**
```bash
# Para banco local
supabase db reset

# Para banco remoto
supabase db push
```

### Erro: "Session expired"

**Causa:** Token expirou e não foi renovado

**Solução:**
- Verificar se `autoRefreshToken: true` está configurado
- Implementar listener de `onAuthStateChange`
- Fazer logout e login novamente

### Erro: "CORS error"

**Causa:** URL não está nas configurações permitidas

**Solução:**
1. Ir em **Authentication > URL Configuration**
2. Adicionar URL do frontend em **Site URL**
3. Adicionar URLs em **Redirect URLs**

## 📊 Estrutura de Migrações SQL

### Exemplo de Migração Inicial

```sql
-- 001_create_initial_schema.sql

-- Criar ENUM para status
CREATE TYPE order_status AS ENUM (
  'pending',
  'processing',
  'shipped',
  'delivered',
  'cancelled'
);

-- Criar tabela
CREATE TABLE IF NOT EXISTS pedidos (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  numero_pedido VARCHAR(20) UNIQUE NOT NULL,
  cliente_id UUID NOT NULL REFERENCES clientes(id),
  status order_status DEFAULT 'pending',
  valor_total DECIMAL(12, 2) NOT NULL DEFAULT 0,
  criado_em TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  atualizado_em TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  criado_por UUID REFERENCES auth.users(id)
);

-- Criar índices
CREATE INDEX idx_pedidos_cliente ON pedidos(cliente_id);
CREATE INDEX idx_pedidos_status ON pedidos(status);
CREATE INDEX idx_pedidos_criado_em ON pedidos(criado_em DESC);

-- Trigger para atualizar timestamp
CREATE OR REPLACE FUNCTION atualizar_timestamp()
RETURNS TRIGGER AS $$
BEGIN
  NEW.atualizado_em = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_atualizar_pedidos_timestamp
  BEFORE UPDATE ON pedidos
  FOR EACH ROW
  EXECUTE FUNCTION atualizar_timestamp();

-- Habilitar RLS
ALTER TABLE pedidos ENABLE ROW LEVEL SECURITY;

-- Políticas RLS
CREATE POLICY "authenticated_read_pedidos"
  ON pedidos FOR SELECT
  TO authenticated
  USING (true);

CREATE POLICY "authenticated_insert_pedidos"
  ON pedidos FOR INSERT
  TO authenticated
  WITH CHECK (true);
```

## 🎓 Conceitos-Chave para Estabilidade

### 1. Gerenciamento de Sessão Eficiente

O sistema de cache de sessão (`getSessionOnce`) é crucial para:
- Evitar múltiplas chamadas simultâneas ao Supabase
- Prevenir race conditions
- Melhorar performance
- Reduzir latência

### 2. Auto-Refresh de Tokens

Configurar `autoRefreshToken: true` garante:
- Usuário permanece logado
- Tokens são renovados automaticamente
- Sem interrupções na experiência do usuário

### 3. Persistência de Sessão

Usar `localStorage` com `persistSession: true`:
- Mantém usuário logado após reload
- Sobrevive a fechamento de abas
- Experiência contínua

### 4. Tipagem TypeScript Completa

Gerar tipos do banco de dados proporciona:
- Autocomplete em todas as queries
- Detecção de erros em tempo de desenvolvimento
- Refatoração segura
- Documentação automática

### 5. React Query para Cache

Integração com React Query oferece:
- Cache automático de queries
- Invalidação inteligente
- Retry automático em falhas
- Loading e error states
